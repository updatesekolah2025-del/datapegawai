/** * GOOGLE APPS SCRIPT - SISTEM DATA GURU * Copy paste script ini ke Google Apps Script Editor * Pastikan Sheet ID sudah benar: 1WSr7jCeBuaEbMOQwU2Jl7e1uuJwlSABFRs4AnoEDQqU */ // Konfigurasi const SHEET_ID = '1WSr7jCeBuaEbMOQwU2Jl7e1uuJwlSABFRs4AnoEDQqU'; const SHEET_NAME = 'Sheet1'; // Header kolom sesuai dengan form aplikasi const HEADERS = [ 'No', 'Timestamp', 'Nama Lengkap', 'NIP', 'NUPTK', 'NIK', 'Tempat Lahir', 'Tanggal Lahir', 'Jenis Kelamin', 'Agama', 'Status Perkawinan', 'Alamat', 'No. Telepon', 'Email', 'Sekolah', 'Status Kepegawaian', 'Jenis PTK', 'Pendidikan Terakhir', 'Jurusan', 'Tahun Lulus', 'Bidang Studi Ijazah', 'Status Sertifikasi', 'No. Sertifikasi', 'Bidang Studi Sertifikasi', 'NRG', 'TMT Sertifikasi', 'TMT Mengajar', 'Mata Pelajaran', 'SK Mengajar', 'TMT SK Mengajar', 'SK GTY', 'TMT SK GTY', 'SK Tugas Pertama', 'TMT SK Tugas Pertama', 'Upload SK Pertama', 'Upload SKBM', 'Upload Sertifikat', 'Tanggal Input' ]; /** * Fungsi untuk menangani POST request (submit form) */ function doPost(e) { try { // Log untuk debugging console.log('POST Request received'); console.log('Request data:', e.postData.contents); // Parse data JSON dari request const data = JSON.parse(e.postData.contents); console.log('Parsed data:', data); // Buka spreadsheet const spreadsheet = SpreadsheetApp.openById(SHEET_ID); const sheet = spreadsheet.getSheetByName(SHEET_NAME); if (!sheet) { throw new Error(`Sheet dengan nama "${SHEET_NAME}" tidak ditemukan`); } // Setup header jika sheet kosong if (sheet.getLastRow() === 0) { console.log('Setting up headers...'); sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]); // Format header const headerRange = sheet.getRange(1, 1, 1, HEADERS.length); headerRange.setBackground('#1e40af'); headerRange.setFontColor('#ffffff'); headerRange.setFontWeight('bold'); headerRange.setHorizontalAlignment('center'); // Freeze header row sheet.setFrozenRows(1); } // Siapkan data untuk baris baru const timestamp = new Date(); const nextRowNumber = sheet.getLastRow(); // Will be the row number after append // Gabungkan SK Tugas Pertama dan Upload berdasarkan status sertifikasi const skTugasPertama = data.is_sertifikasi ? (data.sk_tugas_pertama_sert || '') : (data.sk_tugas_pertama_non || ''); const tmtSkTugasPertama = data.is_sertifikasi ? (data.tmt_sk_tugas_pertama_sert || '') : (data.tmt_sk_tugas_pertama_non || ''); const uploadSkPertama = data.is_sertifikasi ? (data.upload_sk_pertama_sert || '') : (data.upload_sk_pertama_non || ''); const uploadSkbm = data.is_sertifikasi ? (data.upload_skbm_sert || '') : (data.upload_skbm_non || ''); const uploadSertifikat = data.is_sertifikasi ? (data.upload_sertifikat_pendidik || '') : ''; const rowData = [ nextRowNumber, // No urut timestamp, // Timestamp data.nama_lengkap || '', data.nip || '', data.nuptk || '', data.nik || '', data.tempat_lahir || '', data.tanggal_lahir || '', data.jenis_kelamin || '', data.agama || '', data.status_perkawinan || '', data.alamat || '', data.no_telepon || '', data.email || '', data.sekolah || '', data.status_kepegawaian || '', data.jenis_ptk || '', data.pendidikan_terakhir || '', data.jurusan || '', data.tahun_lulus || '', data.bidang_studi_ijazah || '', data.is_sertifikasi ? 'Sudah Sertifikasi' : 'Belum Sertifikasi', data.nomor_sertifikasi || '', data.bidang_studi_sertifikasi || '', data.nrg || '', data.tmt_sertifikasi || '', data.tmt_mengajar || '', data.mapel_diampu || '', data.sk_mengajar || '', data.tmt_sk_mengajar || '', data.sk_gty || '', data.tmt_sk_gty || '', skTugasPertama, tmtSkTugasPertama, uploadSkPertama, uploadSkbm, uploadSertifikat, timestamp.toLocaleDateString('id-ID') + ' ' + timestamp.toLocaleTimeString('id-ID') ]; // Tambahkan baris data ke sheet sheet.appendRow(rowData); console.log('Data berhasil ditambahkan ke sheet'); // Auto-resize kolom untuk readability sheet.autoResizeColumns(1, HEADERS.length); // Response sukses const response = { status: 'success', message: `Data guru ${data.nama_lengkap} berhasil disimpan ke Google Sheets`, timestamp: timestamp.toISOString(), rowNumber: sheet.getLastRow() }; console.log('Response:', response); return ContentService .createTextOutput(JSON.stringify(response)) .setMimeType(ContentService.MimeType.JSON); } catch (error) { console.error('Error in doPost:', error); const errorResponse = { status: 'error', message: `Gagal menyimpan data: ${error.toString()}`, timestamp: new Date().toISOString() }; return ContentService .createTextOutput(JSON.stringify(errorResponse)) .setMimeType(ContentService.MimeType.JSON); } } /** * Fungsi untuk menangani GET request (test koneksi dan ambil data) */ function doGet(e) { try { console.log('GET Request received'); console.log('Parameters:', e.parameter); // Test koneksi if (e.parameter.test === 'connection') { const spreadsheet = SpreadsheetApp.openById(SHEET_ID); const sheet = spreadsheet.getSheetByName(SHEET_NAME); const response = { status: 'success', message: 'Koneksi ke Google Sheets berhasil!', sheetInfo: { sheetId: SHEET_ID, sheetName: SHEET_NAME, totalRows: sheet ? sheet.getLastRow() : 0, totalColumns: sheet ? sheet.getLastColumn() : 0 }, timestamp: new Date().toISOString() }; console.log('Connection test response:', response); return ContentService .createTextOutput(JSON.stringify(response)) .setMimeType(ContentService.MimeType.JSON); } // Ambil data dari sheet untuk auto-sync if (e.parameter.action === 'getData') { const spreadsheet = SpreadsheetApp.openById(SHEET_ID); const sheet = spreadsheet.getSheetByName(SHEET_NAME); if (!sheet || sheet.getLastRow() <= 1) { return ContentService .createTextOutput(JSON.stringify({ status: 'success', message: 'Sheet kosong atau hanya ada header', data: [], timestamp: new Date().toISOString() })) .setMimeType(ContentService.MimeType.JSON); } // Ambil semua data (skip header row) const dataRange = sheet.getRange(2, 1, sheet.getLastRow() - 1, sheet.getLastColumn()); const values = dataRange.getValues(); // Convert ke format JSON const data = values.map(row => { return { no: row[0] || '', timestamp: row[1] || '', nama_lengkap: row[2] || '', nip: row[3] || '', nuptk: row[4] || '', nik: row[5] || '', tempat_lahir: row[6] || '', tanggal_lahir: row[7] || '', jenis_kelamin: row[8] || '', agama: row[9] || '', status_perkawinan: row[10] || '', alamat: row[11] || '', no_telepon: row[12] || '', email: row[13] || '', sekolah: row[14] || '', status_kepegawaian: row[15] || '', jenis_ptk: row[16] || '', pendidikan_terakhir: row[17] || '', jurusan: row[18] || '', tahun_lulus: row[19] || '', bidang_studi_ijazah: row[20] || '', is_sertifikasi: row[21] === 'Sudah Sertifikasi', nomor_sertifikasi: row[22] || '', bidang_studi_sertifikasi: row[23] || '', nrg: row[24] || '', tmt_sertifikasi: row[25] || '', tmt_mengajar: row[26] || '', mapel_diampu: row[27] || '', sk_mengajar: row[28] || '', tmt_sk_mengajar: row[29] || '', sk_gty: row[30] || '', tmt_sk_gty: row[31] || '', sk_tugas_pertama: row[32] || '', tmt_sk_tugas_pertama: row[33] || '', upload_sk_pertama: row[34] || '', upload_skbm: row[35] || '', upload_sertifikat: row[36] || '', created_at: row[37] || new Date().toISOString() }; }).filter(item => item.nama_lengkap); // Filter out empty rows const response = { status: 'success', message: `Berhasil mengambil ${data.length} data guru`, data: data, totalRows: sheet.getLastRow() - 1, timestamp: new Date().toISOString() }; console.log(`Data retrieved: ${data.length} records`); return ContentService .createTextOutput(JSON.stringify(response)) .setMimeType(ContentService.MimeType.JSON); } // Default response untuk GET request lainnya return ContentService .createTextOutput(JSON.stringify({ status: 'info', message: 'Google Apps Script untuk Sistem Data Guru aktif', availableEndpoints: { 'GET ?test=connection': 'Test koneksi ke Google Sheets', 'GET ?action=getData': 'Ambil semua data guru dari sheet', 'POST': 'Submit data guru baru' }, timestamp: new Date().toISOString() })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { console.error('Error in doGet:', error); return ContentService .createTextOutput(JSON.stringify({ status: 'error', message: `Error: ${error.toString()}`, timestamp: new Date().toISOString() })) .setMimeType(ContentService.MimeType.JSON); } } /** * Fungsi helper untuk setup sheet (opsional - bisa dipanggil manual) */ function setupSheet() { try { const spreadsheet = SpreadsheetApp.openById(SHEET_ID); let sheet = spreadsheet.getSheetByName(SHEET_NAME); // Buat sheet baru jika belum ada if (!sheet) { sheet = spreadsheet.insertSheet(SHEET_NAME); } // Clear existing content sheet.clear(); // Setup headers sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]); // Format headers const headerRange = sheet.getRange(1, 1, 1, HEADERS.length); headerRange.setBackground('#1e40af'); headerRange.setFontColor('#ffffff'); headerRange.setFontWeight('bold'); headerRange.setHorizontalAlignment('center'); headerRange.setVerticalAlignment('middle'); // Set column widths sheet.setColumnWidth(1, 50); // No sheet.setColumnWidth(2, 150); // Timestamp sheet.setColumnWidth(3, 200); // Nama Lengkap sheet.setColumnWidth(12, 150); // Alamat sheet.setColumnWidth(28, 200); // Mata Pelajaran // Freeze header row sheet.setFrozenRows(1); console.log('Sheet setup completed successfully'); return 'Sheet berhasil di-setup!'; } catch (error) { console.error('Error in setupSheet:', error); throw error; } }
